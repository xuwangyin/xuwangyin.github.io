\section*{Research Interests}
{\small
\hspace*{1.5em}\textbf{Energy-based generative models}
\begin{itemize}[leftmargin=4em]
  \item Scalable training of energy-based models via adversarial training
  \item Text-to-image generation with energy-based models
  \item Hybrid discriminative-generative models
\end{itemize}
\hspace*{1.5em}\textbf{Machine learning robustness and security}
\begin{itemize}[leftmargin=4em]
  \item Training adversarially robust image models and LLMs
  \item Safety evaluation/red-teaming/jailbreaking LLMs
  \item Detecting adversarial examples
\end{itemize}
