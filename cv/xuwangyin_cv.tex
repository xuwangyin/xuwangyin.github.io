\section*{Research Interests}
{\small
\textbf{Energy-based generative models:} scalable training via adversarial training,
text-to-image generation, and hybrid discriminative-generative models.\par\smallskip
\textbf{Machine learning robustness and security:} adversarially robust image models and LLMs,
safety evaluation and red-teaming of LLMs, and adversarial example detection.\par
}
